## OneInit 扩展功能开发文档

### 一、文档概述

本文档面向 OneInit 核心开发者，详细说明 **环境捕获（`capture`）** 和 **数据迁移（`export/import`）** 两大功能的设计与实现方案。

**设计原则**：
1. **非侵入式检测**：只读扫描，不修改用户现有环境。
2. **渐进式增强**：先支持主流环境（Python/Node/Git），再扩展至其他。
3. **跨平台透明**：对 Windows/Linux/macOS 统一抽象，差异封装在平台适配层。

#### 3.2 检测器 Trait 设计

```rust
// src/capture/detector/mod.rs
use async_trait::async_trait;

/// 环境检测器 Trait
#[async_trait]
pub trait EnvDetector: Send + Sync {
    /// 检测目标环境是否已安装
    async fn detect(&self) -> Result<Option<RuntimeEnv>, DetectorError>;

    /// 返回检测器名称（用于日志）
    fn name(&self) -> &'static str;

    /// 检测优先级（数字越小越优先）
    fn priority(&self) -> u8 { 50 }
}

/// 检测器调度器
pub struct DetectorScheduler {
    detectors: Vec<Box<dyn EnvDetector>>,
}

impl DetectorScheduler {
    pub fn new() -> Self {
        Self { detectors: Vec::new() }
    }

    /// 注册所有检测器
    pub fn register_defaults(&mut self) {
        self.detectors.push(Box::new(PythonDetector::new()));
        self.detectors.push(Box::new(NodeDetector::new()));
        self.detectors.push(Box::new(GitDetector::new()));
        self.detectors.push(Box::new(MysqlDetector::new()));
        // 按优先级排序
        self.detectors.sort_by_key(|d| d.priority());
    }

    /// 执行检测
    pub async fn scan(&self) -> HashMap<String, Option<RuntimeEnv>> {
        let mut results = HashMap::new();
        for detector in &self.detectors {
            let name = detector.name();
            match detector.detect().await {
                Ok(Some(env)) => { results.insert(name.to_string(), Some(env)); }
                Ok(None) => { results.insert(name.to_string(), None); }
                Err(e) => {
                    tracing::warn!("Detector {} failed: {}", name, e);
                    results.insert(name.to_string(), None);
                }
            }
        }
        results
    }
}
```

#### 3.3 Python 检测器实现示例

```rust
// src/capture/detector/python.rs
use super::{EnvDetector, RuntimeEnv};
use std::process::Command;
use std::path::PathBuf;
use regex::Regex;

pub struct PythonDetector;

impl PythonDetector {
    pub fn new() -> Self { Self }

    /// 解析 Python 版本输出
    fn parse_version(output: &str) -> Option<String> {
        let re = Regex::new(r"Python (\d+\.\d+\.\d+)").unwrap();
        re.captures(output).and_then(|cap| cap.get(1).map(|m| m.as_str().to_string()))
    }

    /// 检测 pip 镜像源
    fn detect_pip_mirror() -> Option<String> {
        let output = Command::new("pip")
            .args(["config", "get", "index-url"])
            .output()
            .ok()?;
        if output.status.success() {
            String::from_utf8(output.stdout).ok().map(|s| s.trim().to_string())
        } else {
            None
        }
    }

    /// 获取全局包列表
    fn list_global_packages() -> Vec<String> {
        let output = Command::new("pip")
            .args(["list", "--format=freeze"])
            .output()
            .ok()?;
        if output.status.success() {
            let stdout = String::from_utf8_lossy(&output.stdout);
            stdout.lines()
                .filter(|line| !line.is_empty())
                .map(|s| s.to_string())
                .collect()
        } else {
            Vec::new()
        }
    }
}

#[async_trait::async_trait]
impl EnvDetector for PythonDetector {
    fn name(&self) -> &'static str { "python" }

    async fn detect(&self) -> Result<Option<RuntimeEnv>, DetectorError> {
        // 1. 查找 python3 或 python 命令
        let python_cmd = if cfg!(windows) {
            "python.exe"
        } else {
            "python3"
        };

        let which_output = Command::new(if cfg!(windows) { "where" } else { "which" })
            .arg(python_cmd)
            .output()
            .map_err(|e| DetectorError::CommandNotFound(e.to_string()))?;

        if !which_output.status.success() {
            return Ok(None); // 未安装
        }

        let install_path = String::from_utf8_lossy(&which_output.stdout)
            .lines()
            .next()
            .map(|s| PathBuf::from(s.trim()))
            .ok_or(DetectorError::ParseError("无法解析路径".to_string()))?;

        // 2. 获取版本
        let version_output = Command::new(&install_path)
            .arg("--version")
            .output()
            .map_err(|e| DetectorError::ExecutionFailed(e.to_string()))?;

        let version_str = String::from_utf8_lossy(&version_output.stdout);
        let version = Self::parse_version(&version_str)
            .ok_or(DetectorError::ParseError("无法解析版本号".to_string()))?;

        // 3. 收集配置
        let mut mirrors = HashMap::new();
        if let Some(mirror) = Self::detect_pip_mirror() {
            mirrors.insert("pip".to_string(), mirror);
        }

        let global_packages = Self::list_global_packages();

        // 4. 构建 RuntimeEnv
        let env = RuntimeEnv {
            name: "python".to_string(),
            version,
            install_path,
            mirrors,
            global_packages,
            services: Vec::new(), // Python 无服务
            custom_configs: HashMap::new(),
        };

        Ok(Some(env))
    }
}

#[derive(Debug, thiserror::Error)]
pub enum DetectorError {
    #[error("命令未找到: {0}")]
    CommandNotFound(String),
    #[error("执行失败: {0}")]
    ExecutionFailed(String),
    #[error("解析错误: {0}")]
    ParseError(String),
}
```

```rust
// src/capture/generator.rs
use super::{EnvironmentSnapshot, RuntimeEnv};
use serde_yaml;
use std::fs::File;
use std::io::Write;
use std::path::Path;

pub struct RecipeGenerator;

impl RecipeGenerator {
    /// 从快照生成 YAML
    pub fn generate_yaml(snapshot: &EnvironmentSnapshot) -> Result<String, GeneratorError> {
        let yaml = serde_yaml::to_string(snapshot)
            .map_err(|e| GeneratorError::SerializationFailed(e.to_string()))?;
        Ok(yaml)
    }

    /// 写入文件
    pub fn write_to_file(snapshot: &EnvironmentSnapshot, path: &Path) -> Result<(), GeneratorError> {
        let yaml = Self::generate_yaml(snapshot)?;
        let mut file = File::create(path)
            .map_err(|e| GeneratorError::IoError(e.to_string()))?;
        file.write_all(yaml.as_bytes())
            .map_err(|e| GeneratorError::IoError(e.to_string()))?;
        Ok(())
    }

    /// 生成人类可读的摘要（用于 TUI 展示）
    pub fn generate_summary(snapshot: &EnvironmentSnapshot) -> String {
        let mut summary = String::new();
        summary.push_str("📦 环境摘要:\n\n");
        for (name, env) in &snapshot.envs {
            summary.push_str(&format!("  ✅ {} {}\n", name, env.version));
            if !env.global_packages.is_empty() {
                summary.push_str(&format!("     📚 全局包: {}\n", env.global_packages.join(", ")));
            }
            if !env.mirrors.is_empty() {
                for (key, value) in &env.mirrors {
                    summary.push_str(&format!("     🔄 {} 镜像: {}\n", key, value));
                }
            }
        }
        if !snapshot.dotfiles.is_empty() {
            summary.push_str(&format!("\n  📄 配置文件: {} 个\n", snapshot.dotfiles.len()));
        }
        summary
    }
}

#[derive(Debug, thiserror::Error)]

    #[error("序列化失败: {0}")]
    SerializationFailed(String),
    #[error("IO错误: {0}")]
    IoError(String),
```

#### 4.2 导出打包器

```rust
// src/migration/packer.rs
use super::manifest::{CacheEntry, MigrationManifest, PackageListEntry};
use super::ManifestMetadata;
use crate::capture::{EnvironmentSnapshot, RuntimeEnv};
use flate2::write::GzEncoder;
use flate2::Compression;
use std::fs::File;
use std::io::{BufReader, Read, Write};
use std::path::{Path, PathBuf};
use std::time::{SystemTime, UNIX_EPOCH};
use tar::Builder;
use walkdir::WalkDir;

pub struct Exporter {
    /// 输出路径
    output_path: PathBuf,
    /// 是否包含下载缓存
    include_cache: bool,
    /// 是否包含用户数据（~/.oneinit/envs）
    include_envs: bool,
    /// 压缩算法
    compression: CompressionAlgorithm,
    /// 压缩级别
    compression_level: u8,
}

impl Exporter {
    pub fn new(output_path: PathBuf) -> Self {
        Self {
            output_path,
            include_cache: true,
            include_envs: false,
            compression: CompressionAlgorithm::Gzip,
            compression_level: 6,
        }
    }

    /// 设置选项
    pub fn with_cache(mut self, enable: bool) -> Self {
        self.include_cache = enable;
        self
    }

    pub fn with_envs(mut self, enable: bool) -> Self {
        self.include_envs = enable;
        self
    }

    pub fn with_compression(mut self, algo: CompressionAlgorithm, level: u8) -> Self {
        self.compression = algo;
        self.compression_level = level.clamp(1, 9);
        self
    }

    /// 执行导出
    pub async fn export(
        &self,
        snapshot: &EnvironmentSnapshot,
        dotfiles: Vec<PathBuf>,
    ) -> Result<PathBuf, ExportError> {
        // 1. 创建临时目录
        let temp_dir = tempfile::tempdir()
            .map_err(|e| ExportError::IoError(e.to_string()))?;
        let temp_path = temp_dir.path();

        // 2. 写入 recipe/oneinit.yaml
        let recipe_dir = temp_path.join("recipe");
        std::fs::create_dir(&recipe_dir)
            .map_err(|e| ExportError::IoError(e.to_string()))?;
        let recipe_path = recipe_dir.join("oneinit.yaml");
        crate::capture::RecipeGenerator::write_to_file(snapshot, &recipe_path)
            .map_err(|e| ExportError::RecipeError(e.to_string()))?;

        // 3. 收集缓存文件
        let mut cache_entries = Vec::new();
        if self.include_cache {
            let cache_dir = dirs::cache_dir()
                .unwrap_or_else(|| PathBuf::from(".oneinit/cache"))
                .join("oneinit");
            if cache_dir.exists() {
                let cache_dest = temp_path.join("cache");
                std::fs::create_dir(&cache_dest)
                    .map_err(|e| ExportError::IoError(e.to_string()))?;
                self.copy_cache_files(&cache_dir, &cache_dest, &mut cache_entries)?;
            }
        }

        // 4. 复制 dotfiles
        let dotfile_dest = temp_path.join("dotfiles");
        std::fs::create_dir(&dotfile_dest)
            .map_err(|e| ExportError::IoError(e.to_string()))?;
        let mut dotfile_entries = Vec::new();
        for src in dotfiles {
            let filename = src.file_name()
                .ok_or(ExportError::InvalidPath(src.clone()))?;
            let dest = dotfile_dest.join(filename);
            std::fs::copy(&src, &dest)
                .map_err(|e| ExportError::IoError(e.to_string()))?;
            dotfile_entries.push(dest);
        }

        // 5. 收集全局包列表
        let mut package_lists = Vec::new();
        if let Some(python_env) = snapshot.envs.get("python") {
            if !python_env.global_packages.is_empty() {
                package_lists.push(PackageListEntry {
                    manager: "pip".to_string(),
                    packages: python_env.global_packages.clone(),
                });
            }
        }
        if let Some(node_env) = snapshot.envs.get("nodejs") {
            if !node_env.global_packages.is_empty() {
                package_lists.push(PackageListEntry {
                    manager: "npm".to_string(),
                    packages: node_env.global_packages.clone(),
                });
            }
        }

        // 6. 生成清单（manifest.json）
        let manifest = MigrationManifest {
            metadata: ManifestMetadata {
                tool: "OneInit".to_string(),
                version: env!("CARGO_PKG_VERSION").to_string(),
                created_at: SystemTime::now()
                    .duration_since(UNIX_EPOCH)
                    .unwrap_or_default()
                    .as_secs(),
                source_os: std::env::consts::OS.to_string(),
                source_hostname: gethostname::gethostname()
                    .to_string_lossy()
                    .to_string(),
                compression: format!("{:?}", self.compression),
                total_size: 0, // 稍后计算
            },
            recipe: PathBuf::from("recipe/oneinit.yaml"),
            cache_files: cache_entries,
            dotfiles: dotfile_entries.into_iter()
                .map(|p| p.to_string_lossy().to_string())
                .collect(),
            global_packages: package_lists,
            checksums: HashMap::new(),
        };

        // 7. 写入 manifest.json
        let manifest_json = serde_json::to_string_pretty(&manifest)
            .map_err(|e| ExportError::SerializationError(e.to_string()))?;
        std::fs::write(temp_path.join("manifest.json"), manifest_json)
            .map_err(|e| ExportError::IoError(e.to_string()))?;

        // 8. 打包为 tar.gz
        self.create_archive(temp_path, &self.output_path)?;

        Ok(self.output_path.clone())
    }

    /// 复制缓存文件并计算元信息
    fn copy_cache_files(
        &self,
        src_dir: &Path,
        dest_dir: &Path,
        entries: &mut Vec<CacheEntry>,
    ) -> Result<(), ExportError> {
        for entry in WalkDir::new(src_dir)
            .into_iter()
            .filter_map(|e| e.ok())
            .filter(|e| e.file_type().is_file())
        {
            let src_path = entry.path();
            let filename = src_path.file_name()
                .ok_or(ExportError::InvalidPath(src_path.to_path_buf()))?;
            let dest_path = dest_dir.join(filename);

            // 计算 SHA256
            let sha256 = Self::calculate_sha256(src_path)?;

            // 获取大小
            let size = std::fs::metadata(src_path)
                .map_err(|e| ExportError::IoError(e.to_string()))?
                .len();

            // 提取包名（假设文件名格式为 package-version.ext）
            let package_name = filename.to_string_lossy()
                .split('-')
                .next()
                .unwrap_or("unknown")
                .to_string();

            entries.push(CacheEntry {
                package: package_name,
                version: "".to_string(), // 实际应从文件名解析
                filename: filename.to_string_lossy().to_string(),
                size,
                sha256,
            });

            std::fs::copy(src_path, dest_path)
                .map_err(|e| ExportError::IoError(e.to_string()))?;
        }
        Ok(())
    }

    /// 计算文件 SHA256
    fn calculate_sha256(path: &Path) -> Result<String, ExportError> {
        use sha2::{Sha256, Digest};
        let mut file = File::open(path)
            .map_err(|e| ExportError::IoError(e.to_string()))?;
        let mut hasher = Sha256::new();
        let mut buffer = [0; 8192];
        loop {
            let bytes_read = file.read(&mut buffer)
                .map_err(|e| ExportError::IoError(e.to_string()))?;
            if bytes_read == 0 { break; }
            hasher.update(&buffer[..bytes_read]);
        }
        Ok(format!("{:x}", hasher.finalize()))
    }

    /// 创建 tar 归档
    fn create_archive(&self, source_dir: &Path, output_path: &Path) -> Result<(), ExportError> {
        let file = File::create(output_path)
            .map_err(|e| ExportError::IoError(e.to_string()))?;
        let gz_encoder = GzEncoder::new(file, Compression::new(self.compression_level as u32));
        let mut tar_builder = Builder::new(gz_encoder);

        // 递归添加所有文件
        for entry in WalkDir::new(source_dir).into_iter().filter_map(|e| e.ok()) {
            let path = entry.path();
            let relative_path = path.strip_prefix(source_dir)
                .map_err(|e| ExportError::IoError(e.to_string()))?;
            tar_builder.append_path_with_name(path, relative_path)
                .map_err(|e| ExportError::IoError(e.to_string()))?;
        }

        tar_builder.into_inner()
            .map_err(|e| ExportError::IoError(e.to_string()))?;
        Ok(())
    }
}

#[derive(Debug, Clone, Copy)]
pub enum CompressionAlgorithm {
    Gzip,
    Zstd,
}

#[derive(Debug, thiserror::Error)]
pub enum ExportError {
    #[error("IO 错误: {0}")]
    IoError(String),
    #[error("序列化错误: {0}")]
    SerializationError(String),
    #[error("配方生成错误: {0}")]
    RecipeError(String),
    #[error("无效路径: {0:?}")]
    InvalidPath(PathBuf),
}
```

```rust
    archive_path: PathBuf,
    dry_run: bool,
    force_overwrite: bool,
    skip_checksum: bool,
}

impl Importer {
    pub fn new(archive_path: PathBuf) -> Self {
        Self {
            archive_path,
            dry_run: false,
            force_overwrite: false,
            skip_checksum: false,
        }
    }

    pub fn with_dry_run(mut self, enable: bool) -> Self { self.dry_run = enable; self }
    pub fn with_force(mut self, enable: bool) -> Self { self.force_overwrite = enable; self }
    pub fn with_skip_checksum(mut self, enable: bool) -> Self { self.skip_checksum = enable; self }

    /// 执行导入
    pub async fn import(&self) -> Result<MigrationResult, ImportError> {
        // 1. 解压到临时目录
        let temp_dir = tempfile::tempdir()
            .map_err(|e| ImportError::IoError(e.to_string()))?;
        let temp_path = temp_dir.path();

        self.extract_archive(&self.archive_path, temp_path)?;

        // 2. 读取清单
        let manifest_path = temp_path.join("manifest.json");
        let manifest_content = std::fs::read_to_string(&manifest_path)
            .map_err(|e| ImportError::IoError(e.to_string()))?;
        let manifest: MigrationManifest = serde_json::from_str(&manifest_content)
            .map_err(|e| ImportError::InvalidManifest(e.to_string()))?;

        // 3. 验证 SHA256（可选）
        if !self.skip_checksum {
            self.verify_checksums(temp_path, &manifest)?;
        }

        // 4. 恢复配方到 ~/.oneinit/recipes/
        let recipe_src = temp_path.join(&manifest.recipe);
        let recipe_dest = dirs::home_dir()
            .ok_or(ImportError::HomeDirNotFound)?
            .join(".oneinit")
            .join("recipes")
            .join("imported.yaml");
        if !self.dry_run {
            std::fs::create_dir_all(recipe_dest.parent().unwrap())?;
            std::fs::copy(&recipe_src, &recipe_dest)?;
        }

        // 5. 恢复缓存
        let cache_src = temp_path.join("cache");
        if cache_src.exists() {
            let cache_dest = dirs::cache_dir()
                .ok_or(ImportError::CacheDirNotFound)?
                .join("oneinit");
            if !self.dry_run {
                std::fs::create_dir_all(&cache_dest)?;
                self.copy_dir(&cache_src, &cache_dest)?;
            }
        }

        // 6. 恢复 dotfiles
        let dotfiles_src = temp_path.join("dotfiles");
        if dotfiles_src.exists() {
            if !self.dry_run {
                for entry in manifest.dotfiles {
                    let src = dotfiles_src.join(Path::new(&entry));
                    let target = dirs::home_dir()
                        .ok_or(ImportError::HomeDirNotFound)?
                        .join(entry.file_name().unwrap());
                    if target.exists() && !self.force_overwrite {
                        tracing::warn!("跳过已存在的文件: {:?}", target);
                        continue;
                    }
                    std::fs::copy(&src, &target)?;
                }
            }
        }

        // 7. 恢复全局包
        if !self.dry_run {
            self.restore_global_packages(&manifest, &mut |pkg, total| {
                tracing::info!("恢复包 {}/{}", pkg, total);
            })?;
        }

        Ok(MigrationResult {
            recipe_path: recipe_dest,
            cache_restored: manifest.cache_files.len(),
            dotfiles_restored: manifest.dotfiles.len(),
            packages_restored: manifest.global_packages.iter().map(|p| p.packages.len()).sum(),
        })
    }

    fn extract_archive(&self, archive_path: &Path, dest: &Path) -> Result<(), ImportError> {
        let file = File::open(archive_path)
            .map_err(|e| ImportError::IoError(e.to_string()))?;
        let gz = GzDecoder::new(file);
        let mut archive = Archive::new(gz);
        archive.unpack(dest)
            .map_err(|e| ImportError::IoError(e.to_string()))?;
        Ok(())
    }

    fn verify_checksums(&self, root: &Path, manifest: &MigrationManifest) -> Result<(), ImportError> {
        for entry in &manifest.cache_files {
            let file_path = root.join("cache").join(&entry.filename);
            if file_path.exists() {
                let actual = crate::migration::packer::Exporter::calculate_sha256(&file_path)
                    .map_err(|e| ImportError::ChecksumError(e.to_string()))?;
                if actual != entry.sha256 {
                    return Err(ImportError::ChecksumMismatch {
                        file: entry.filename.clone(),
                        expected: entry.sha256.clone(),
                        actual,
                    });
                }
            }
        }
        Ok(())
    }

    fn copy_dir(&self, src: &Path, dest: &Path) -> Result<(), ImportError> {
        for entry in walkdir::WalkDir::new(src).into_iter().filter_map(|e| e.ok()) {
            let rel_path = entry.path().strip_prefix(src)
                .map_err(|e| ImportError::IoError(e.to_string()))?;
            let dest_path = dest.join(rel_path);
            if entry.file_type().is_dir() {
                std::fs::create_dir_all(&dest_path)?;
            } else {
                std::fs::copy(entry.path(), &dest_path)?;
            }
        }
        Ok(())
    }

    fn restore_global_packages<F>(&self, manifest: &MigrationManifest, progress: F) -> Result<(), ImportError>
    where
        F: Fn(usize, usize),
    {
        for list in &manifest.global_packages {
            match list.manager.as_str() {
                "pip" => {
                    for (idx, pkg) in list.packages.iter().enumerate() {
                        tracing::info!("安装 pip 包: {}", pkg);
                        let status = std::process::Command::new("pip")
                            .args(["install", pkg])
                            .status()
                            .map_err(|e| ImportError::IoError(e.to_string()))?;
                        if !status.success() {
                            tracing::warn!("pip 安装失败: {}", pkg);
                        }
                        progress(idx + 1, list.packages.len());
                    }
                }
                "npm" => {
                    // 类似实现
                }
                _ => tracing::warn!("未知包管理器: {}", list.manager),
            }
        }
        Ok(())
    }
}

pub struct MigrationResult {
    pub recipe_path: PathBuf,
    pub cache_restored: usize,
    pub dotfiles_restored: usize,
    pub packages_restored: usize,
}

#[derive(Debug, thiserror::Error)]
pub enum ImportError {
    #[error("IO 错误: {0}")]
    IoError(String),
    #[error("清单无效: {0}")]
    InvalidManifest(String),
    #[error("校验和不匹配: {file} 期望 {expected} 实际 {actual}")]
    ChecksumMismatch { file: String, expected: String, actual: String },
    #[error("校验和错误: {0}")]
    ChecksumError(String),
    #[error("用户目录未找到")]
    HomeDirNotFound,
    #[error("缓存目录未找到")]
    CacheDirNotFound,
}
```

#### 5.1 平台差异汇总

| 操作 | Windows | Linux | macOS |
| :--- | :--- | :--- | :--- |
| **查找可执行文件** | `where python.exe` | `which python3` | `which python3` / `brew --prefix` |
| **常见安装路径** | `C:\Python3x`、`%LOCALAPPDATA%\Programs\Python` | `/usr/bin`、`/usr/local/bin` | `/usr/local/bin`、`/opt/homebrew/bin` |
| **配置文件位置** | `%USERPROFILE%\.bashrc` (Git Bash) | `~/.bashrc`、`~/.zshrc` | `~/.zshrc` (默认) |
| **pip 配置** | `%APPDATA%\pip\pip.ini` | `~/.pip/pip.conf` | `~/.pip/pip.conf` |
| **npm 配置** | `%USERPROFILE%\.npmrc` | `~/.npmrc` | `~/.npmrc` |
| **注册表访问** | 支持（`winreg` crate） | 无 | 无 |
| **路径分隔符** | `\` | `/` | `/` |
| **环境变量分隔符** | `;` | `:` | `:` |

```rust
// src/platform/mod.rs
use std::path::PathBuf;

/// 平台特定路径解析器
pub trait PlatformPathResolver {
    /// 查找命令的完整路径
    fn find_command(&self, name: &str) -> Option<PathBuf>;

    /// 获取常见安装目录
    fn common_install_dirs(&self) -> Vec<PathBuf>;

    /// 获取配置文件目录
    fn config_dirs(&self) -> Vec<PathBuf>;
}

#[cfg(windows)]
mod windows {
    use super::*;
    use winreg::RegKey;
    use winreg::enums::*;

    pub struct WindowsResolver;

    impl PlatformPathResolver for WindowsResolver {
        fn find_command(&self, name: &str) -> Option<PathBuf> {
            // 使用 where 命令
            let output = std::process::Command::new("where")
                .arg(name)
                .output()
                .ok()?;
            if output.status.success() {
                String::from_utf8(output.stdout)
                    .ok()
                    .and_then(|s| s.lines().next().map(PathBuf::from))
            } else {
                None
            }
        }

        fn common_install_dirs(&self) -> Vec<PathBuf> {
            let mut dirs = Vec::new();
            // 从注册表读取 Python 安装路径
            if let Ok(hklm) = RegKey::predef(HKEY_LOCAL_MACHINE)
                .open_subkey(r"SOFTWARE\Python\PythonCore")
            {
                for key in hklm.enum_keys().filter_map(|k| k.ok()) {
                    if let Ok(sub) = hklm.open_subkey(&key) {
                        if let Ok(install_path) = sub.get_value::<String, _>("InstallPath") {
                            dirs.push(PathBuf::from(install_path));
                        }
                    }
                }
            }
            // Program Files
            if let Ok(program_files) = std::env::var("ProgramFiles") {
                dirs.push(PathBuf::from(program_files).join("Python"));
            }
            // 用户安装
            if let Ok(local_appdata) = std::env::var("LOCALAPPDATA") {
                dirs.push(PathBuf::from(local_appdata).join("Programs").join("Python"));
            }
            dirs
        }

        fn config_dirs(&self) -> Vec<PathBuf> {
            let mut dirs = Vec::new();
            if let Ok(appdata) = std::env::var("APPDATA") {
                dirs.push(PathBuf::from(appdata).join("pip"));
                dirs.push(PathBuf::from(appdata));
            }
            if let Ok(userprofile) = std::env::var("USERPROFILE") {
                dirs.push(PathBuf::from(userprofile));
            }
            dirs
        }
    }
}

#[cfg(any(target_os = "linux", target_os = "macos"))]
mod unix {
    use super::*;

    pub struct UnixResolver;

    impl PlatformPathResolver for UnixResolver {
        fn find_command(&self, name: &str) -> Option<PathBuf> {
            let output = std::process::Command::new("which")
                .arg(name)
                .output()
                .ok()?;
            if output.status.success() {
                String::from_utf8(output.stdout)
                    .ok()
                    .map(|s| PathBuf::from(s.trim()))
            } else {
                None
            }
        }

        fn common_install_dirs(&self) -> Vec<PathBuf> {
            vec![
                PathBuf::from("/usr/bin"),
                PathBuf::from("/usr/local/bin"),
                PathBuf::from("/opt"),
            ]
        }

        fn config_dirs(&self) -> Vec<PathBuf> {
            let home = dirs::home_dir().unwrap_or_else(|| PathBuf::from("/"));
            vec![
                home.join(".pip"),
                home.join(".npm"),
                home,
            ]
        }
    }
}

/// 获取当前平台的解析器（工厂方法）
pub fn get_resolver() -> Box<dyn PlatformPathResolver> {
    #[cfg(windows)] {
        Box::new(windows::WindowsResolver)
    }
    #[cfg(any(target_os = "linux", target_os = "macos"))] {
        Box::new(unix::UnixResolver)
    }
```

```rust
        RecipeGenerator::write_to_file(&snapshot, Path::new(&args.output))?;
        println!("✅ 配方已生成: {}", args.output);
    }
    Ok(())
}
```
